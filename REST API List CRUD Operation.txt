	function AddListItem()  
{  
    var siglum = "Test Siglum";
    
    $.ajax  
        ({  
        url: _spPageContextInfo.webAbsoluteUrl + "/_api/web/lists/GetByTitle('Siglum')/items",  
        type: "POST",  
        data: JSON.stringify  
        ({  
            __metadata:  
            {  
                type: "SP.Data.SiglumListItem"  
            },  
            Title: siglum,  
			
           
        }),  
        headers:  
        {  
            "Accept": "application/json;odata=verbose",  
            "Content-Type": "application/json;odata=verbose",  
            "X-RequestDigest": $("#__REQUESTDIGEST").val(),  
            "X-HTTP-Method": "POST"  
        },  
        success: function(data, status, xhr)  
        {  
           console.log('Item added success'); 
        },  
        error: function(xhr, status, error)  
        {  
		  console.log('Item added failed');
           
        }  
    });  
}  



function deleteListItem()  
{  
    $.ajax  
    ({  
        url: _spPageContextInfo.webAbsoluteUrl + "/_api/web/lists/GetByTitle('Siglum')/items(1)",  
        type: "POST",  
        headers:  
        {  
            "X-RequestDigest": $("#__REQUESTDIGEST").val(),  
            "IF-MATCH": "*",  
            "X-HTTP-Method": "DELETE"  
        },  
        success: function(data, status, xhr)  
        {  
          console.log("Item deleted success");
        },  
        error: function(xhr, status, error)  
        {  
		 console.log("Item deleted failed");
            //$("#ResultDiv").empty().text(data.responseJSON.error);  
        }  
    });  
}  


function updateListItem()  
{  
    var siglum = "updated value";  
    $.ajax  
    ({  
        url: _spPageContextInfo.webAbsoluteUrl + "/_api/web/lists/GetByTitle('Siglum')/items(2)", // list item ID  
        type: "POST",  
        data: JSON.stringify  
        ({  
            __metadata:  
            {  
                type: "SP.Data.SiglumListItem"  
            },  
            Title: siglum  
        }),  
        headers:  
        {  
            "Accept": "application/json;odata=verbose",  
            "Content-Type": "application/json;odata=verbose",  
            "X-RequestDigest": $("#__REQUESTDIGEST").val(),  
            "IF-MATCH": "*",  
            "X-HTTP-Method": "MERGE"  
        },  
        success: function(data, status, xhr)  
        {  
           console.log('Item updated');
        },  
        error: function(xhr, status, error)  
        {  console.log('Item update failed');
           
        }  
    });  
}
