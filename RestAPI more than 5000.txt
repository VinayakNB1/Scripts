<script>

} );

var url = _spPageContextInfo.webAbsoluteUrl + "/_api/web/lists/getbytitle('mylist')/items?$select=Title&$top=5000";
var listData =[];

function retrieveListItems()
{
        $.ajax({
                   url: url,
                   method: "GET",
                   headers: { "Accept": "application/json; odata=verbose" },
                   success: function (data) {
				  			   			   
				   listData.push(data.d.results);
				
		if(data.d.__next)
		{
			url = data.d.__next;
			retrieveListItems();
		}
		
		if(data.d.__next == undefined)
		{
		console.log(listData.length);	
		for(var i=0;i<listData.length;i++)
		{
			for(var j=0;j<listData[i].length;j++)
			{
				$('#tablebody').append("<tr><td>" + listData[i][j].Title + "</td><td>" + listData[i][j].Title + "</td><td>" +listData[i][j].Title + "</td>");
			}
		}
		
			
		}
		
		
					        
                  },
                  error: function (data) {
                      alert("Error: "+ data);
                 }
          
    });

}

function createTable()
{
	    $('#example').dataTable( {
        // Sets the row-num-selection "Show %n entries" for the user
        "lengthMenu": [ 20, 40, 60, 80, 100 ], 
        
        // Set the default no. of rows to display
        "pageLength": 20 
    } );
	
}
</script>
